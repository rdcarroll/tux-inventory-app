select i.Barcode, k.ModelName, i.Size, s.Name, k.cost
from inventory_items i
inner join inventory k on i.InventoryID = k.ID
left join status_types s on i.StatusID = s.ID
where s.name not in ( 'Available','Sold', 'Laundry', 'Damaged','Northshore' );
